@echo off
rem RomSync: conecta este PC con la tablet. Lo genero la tablet para ella misma (trae su direccion y un codigo).
net session >nul 2>&1 || (powershell -NoProfile -Command "Start-Process -Verb RunAs -FilePath '%~f0'" & exit /b)
chcp 65001 >nul
powershell -NoProfile -ExecutionPolicy Bypass -Command "$t = Get-Content -LiteralPath '%~f0' -Encoding UTF8; $i = [array]::IndexOf($t, '#PS'); iex ($t[($i + 1)..($t.Count - 1)] -join [char]10)"
pause
exit /b
#PS
$tablet = '__TABLET__'
$token = '__TOKEN__'
$known = @('psx','ps2','ps3','psp','psvita','snes','nes','gb','gba','gbc','nds','n3ds','3ds','n64','gc','wii','wiiu','switch','genesis','megadrive','dreamcast','saturn','mastersystem','gamegear','arcade','mame','fbneo','neogeo','pcengine','segacd','atari2600')
function Paso($n, $texto) { Write-Host ''; Write-Host ("  " + $n + ". " + $texto) -ForegroundColor Cyan }

Write-Host ''
Write-Host '  RomSync - conectar este PC con la tablet' -ForegroundColor Cyan

try {
    Paso 1 'Activando SFTP (el servidor OpenSSH de Windows)...'
    $cap = Get-WindowsCapability -Online -Name 'OpenSSH.Server*' | Select-Object -First 1
    if ($cap.State -ne 'Installed') {
        Write-Host '     Instalando (puede tardar unos minutos)...'
        Add-WindowsCapability -Online -Name $cap.Name | Out-Null
    }
    Set-Service -Name sshd -StartupType Automatic
    Start-Service sshd
    if (-not (Get-NetFirewallRule -Name 'OpenSSH-Server-In-TCP' -ErrorAction SilentlyContinue)) {
        New-NetFirewallRule -Name 'OpenSSH-Server-In-TCP' -DisplayName 'OpenSSH (RomSync)' -Enabled True -Direction Inbound -Protocol TCP -Action Allow -LocalPort 22 | Out-Null
    } else { Enable-NetFirewallRule -Name 'OpenSSH-Server-In-TCP' }
    Write-Host '     Listo.' -ForegroundColor Green
} catch {
    Write-Host ('     No se pudo activar SFTP: ' + $_.Exception.Message) -ForegroundColor Red
    Write-Host '     Prueba a mano: Configuracion -> Sistema -> Caracteristicas opcionales -> Servidor OpenSSH.'
    return
}

# La IP de este PC en la misma red que la tablet.
$net = ($tablet -split '\.')[0..2] -join '.'
$ips = @(Get-NetIPAddress -AddressFamily IPv4 | Where-Object { $_.IPAddress -like "$net.*" } | ForEach-Object { $_.IPAddress })
if ($ips.Count -eq 0) { $ips = @(Get-NetIPAddress -AddressFamily IPv4 | Where-Object { $_.IPAddress -match '^(192\.168\.|10\.|172\.)' } | ForEach-Object { $_.IPAddress }) }
$ip = $ips | Select-Object -First 1

Paso 2 'Buscando tu carpeta de ROMs...'
$cands = @()
foreach ($d in (Get-PSDrive -PSProvider FileSystem | Where-Object { $_.Free -ne $null })) {
    $dirs = @(Get-ChildItem -LiteralPath $d.Root -Directory -Depth 2 -ErrorAction SilentlyContinue | Where-Object { $_.Name -match '^roms?$' })
    foreach ($r in $dirs) {
        $subs = @(Get-ChildItem -LiteralPath $r.FullName -Directory -ErrorAction SilentlyContinue | Where-Object { $known -contains $_.Name.ToLower() })
        if ($subs.Count -ge 2) { $cands += [pscustomobject]@{ Path = $r.FullName; N = $subs.Count } }
    }
}
$roms = $null
if ($cands.Count -gt 0) {
    $i = 1
    foreach ($c in $cands) { Write-Host ("     [" + $i + "] " + $c.Path + "  (" + $c.N + " plataformas)"); $i++ }
    Write-Host '     [0] Elegir otra carpeta'
    $sel = Read-Host '     Numero de tu carpeta de ROMs (Enter = 1)'
    if ($sel -eq '') { $sel = '1' }
    if ($sel -ne '0' -and [int]$sel -ge 1 -and [int]$sel -le $cands.Count) { $roms = $cands[[int]$sel - 1].Path }
}
if (-not $roms) {
    Add-Type -AssemblyName System.Windows.Forms
    $dlg = New-Object System.Windows.Forms.FolderBrowserDialog
    $dlg.Description = 'Elige la carpeta de ROMs (la que tiene una carpeta por plataforma: psx, ps2, switch...)'
    if ($dlg.ShowDialog() -ne 'OK') { Write-Host '     Cancelado.'; return }
    $roms = $dlg.SelectedPath
}
Write-Host ('     ROMs: ' + $roms) -ForegroundColor Green

Paso 3 ('Tu usuario es ' + $env:USERNAME + '. Escribe la contrasena de Windows.')
Write-Host '     (La de tu cuenta, no el PIN. Si entras con cuenta de Microsoft, es la de esa cuenta.)'
for ($try = 1; $try -le 3; $try++) {
    $sec = Read-Host '     Contrasena' -AsSecureString
    $pass = [Runtime.InteropServices.Marshal]::PtrToStringAuto([Runtime.InteropServices.Marshal]::SecureStringToBSTR($sec))
    $body = @{ host = $ip; port = 22; user = $env:USERNAME; password = $pass; roms = $roms } | ConvertTo-Json
    Write-Host '     Probando desde la tablet...'
    try {
        $r = Invoke-RestMethod -Method Post -Uri ("http://" + $tablet + "/api/pair?t=" + $token) -Body ([Text.Encoding]::UTF8.GetBytes($body)) -ContentType 'application/json; charset=utf-8' -TimeoutSec 60
        Write-Host ''
        Write-Host ('  Listo: ' + $r.message) -ForegroundColor Green
        Write-Host '  Ya puedes cerrar esta ventana. La tienda de la tablet se actualiza sola.'
        return
    } catch {
        $msg = $_.ErrorDetails.Message
        try { $msg = ($msg | ConvertFrom-Json).error } catch {}
        if (-not $msg) { $msg = $_.Exception.Message }
        Write-Host ('     La tablet dice: ' + $msg) -ForegroundColor Red
        if ($msg -notmatch 'contrase') { return }
    }
}
